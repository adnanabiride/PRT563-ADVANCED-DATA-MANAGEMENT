//
// Conventions
//   Labels       : PascalCase nouns             (Person, CourseOffering)
//   Rel. types   : UPPER_SNAKE_CASE verbs       (LEADS, TOOK, HOLDS_ACCOUNT)
//   Properties   : snake_case, identical to the source column names, so every
//                  property can be traced back to its relational column.
//   Empty CSV fields are guarded with CASE ... END so that an empty value is
//   never stored as an empty string (a null property is simply not written).

// -----------------------------------------------------------------------------
// 0. OPTIONAL RESET - uncomment only when re-running into a non-empty database
// -----------------------------------------------------------------------------

// -----------------------------------------------------------------------------
// 1. CONSTRAINTS
//    Every PRIMARY KEY and UNIQUE constraint of the relational schema is
//    re-created as a uniqueness constraint (Community Edition). Each is backed
//    automatically by a range index, so every MATCH/MERGE on a key below is an
//    index seek rather than a label scan. Composite constraints preserve the
//    composite keys of the reified relationships (course offering, enrolment,
//    placement, invoice line). Rules that Community Edition cannot declare
//    (NOT NULL, relationship cardinality) are checked in section 8.
// -----------------------------------------------------------------------------
CREATE CONSTRAINT person_id_unique IF NOT EXISTS
FOR (p:Person) REQUIRE p.person_id IS UNIQUE;

CREATE CONSTRAINT person_email_unique IF NOT EXISTS
FOR (p:Person) REQUIRE p.email IS UNIQUE;

CREATE CONSTRAINT student_no_unique IF NOT EXISTS
FOR (s:Student) REQUIRE s.student_no IS UNIQUE;

CREATE CONSTRAINT student_usi_unique IF NOT EXISTS

CREATE CONSTRAINT staff_no_unique IF NOT EXISTS

CREATE CONSTRAINT campus_name_unique IF NOT EXISTS

CREATE CONSTRAINT term_code_unique IF NOT EXISTS
FOR (t:Term) REQUIRE t.term_code IS UNIQUE;

CREATE CONSTRAINT term_year_number_unique IF NOT EXISTS
FOR (t:Term) REQUIRE (t.academic_year, t.term_number) IS UNIQUE;

CREATE CONSTRAINT program_code_unique IF NOT EXISTS
FOR (p:Program) REQUIRE p.program_code IS UNIQUE;

CREATE CONSTRAINT program_name_unique IF NOT EXISTS

CREATE CONSTRAINT enrolment_key_unique IF NOT EXISTS

CREATE CONSTRAINT qualification_id_unique IF NOT EXISTS

CREATE CONSTRAINT certificate_no_unique IF NOT EXISTS

CREATE CONSTRAINT placement_key_unique IF NOT EXISTS

CREATE CONSTRAINT account_no_unique IF NOT EXISTS
FOR (b:BillingAccount) REQUIRE b.account_no IS UNIQUE;

CREATE CONSTRAINT invoice_no_unique IF NOT EXISTS
FOR (i:Invoice) REQUIRE i.invoice_no IS UNIQUE;

CREATE CONSTRAINT invoice_line_key_unique IF NOT EXISTS

CREATE CONSTRAINT payment_id_unique IF NOT EXISTS
FOR (p:Payment) REQUIRE p.payment_id IS UNIQUE;

// Category rule (Assessment 2 feedback): a billing account belongs to ONE
// holder only - a student, a guardian or a host employer. account_no is copied
// onto [:HOLDS_ACCOUNT] so that a relationship property uniqueness constraint
// (Neo4j 5.7+, available in Community Edition) rejects any second holder,
// whatever its label. The relational version needed three UNIQUE indexes and
// six triggers for the same rule (supplementary/a2_feedback_fixes.sql).
CREATE CONSTRAINT holds_account_unique IF NOT EXISTS
FOR ()-[h:HOLDS_ACCOUNT]-() REQUIRE h.account_no IS UNIQUE;


// -----------------------------------------------------------------------------
// 2. QUERY-DRIVEN INDEXES
//    Created only where a use case in queries.txt filters or sorts on a
//    non-key property. Relationship property indexes (Neo4j 5) let the
//    engine seek TOOK relationships by status or outcome directly instead of
//    expanding every student's enrolments.
// -----------------------------------------------------------------------------
CREATE INDEX staff_blue_card_expiry IF NOT EXISTS
FOR (s:Staff) ON (s.blue_card_expiry);

CREATE INDEX enrolment_status IF NOT EXISTS
FOR (e:Enrolment) ON (e.enrolment_status);

CREATE INDEX invoice_status_due IF NOT EXISTS
FOR (i:Invoice) ON (i.invoice_status, i.due_date);

CREATE INDEX took_status IF NOT EXISTS
FOR ()-[r:TOOK]-() ON (r.status);

CREATE INDEX took_competency_outcome IF NOT EXISTS
FOR ()-[r:TOOK]-() ON (r.competency_outcome);

// 6.1 GUARDIANSHIP (junction table) -> [:GUARDIAN_OF] with its attributes
LOAD CSV WITH HEADERS FROM 'file:///guardianship.csv' AS row
MATCH (g:Guardian {person_id: row.guardian_person_id})
MATCH (s:Student  {person_id: row.student_person_id})
MERGE (g)-[r:GUARDIAN_OF]->(s)
SET r.relationship_type  = row.relationship_type,
    r.is_primary_contact = (row.is_primary_contact = 'Y');

// 6.2 STAFF.supervisor_person_id (unary relationship) -> [:SUPERVISES]
LOAD CSV WITH HEADERS FROM 'file:///staff.csv' AS row
WITH row WHERE row.supervisor_person_id <> ''
MATCH (boss:Staff {person_id: row.supervisor_person_id})
MATCH (s:Staff    {person_id: row.person_id})
MERGE (boss)-[:SUPERVISES]->(s);

// 6.3 PROGRAM_SUBJECT (junction table) -> [:INCLUDES]
LOAD CSV WITH HEADERS FROM 'file:///program_subject.csv' AS row
MATCH (p:Program {program_code: row.program_code})
MATCH (s:Subject {subject_code: row.subject_code})
MERGE (p)-[r:INCLUDES]->(s)
SET r.is_core     = (row.is_core = 'Y'),
    r.sequence_no = toInteger(row.sequence_no);

// 6.4 SUBJECT_PREREQUISITE (recursive junction) -> [:REQUIRES]
LOAD CSV WITH HEADERS FROM 'file:///subject_prerequisite.csv' AS row
MATCH (s:Subject   {subject_code: row.subject_code})
MATCH (pre:Subject {subject_code: row.prerequisite_code})
MERGE (s)-[:REQUIRES]->(pre);

// 6.5 COURSE_OFFERING foreign keys -> the three legs of the reified ternary,
//     plus the aggregation's lead trainer
LOAD CSV WITH HEADERS FROM 'file:///course_offering.csv' AS row
MATCH (o:CourseOffering {subject_code: row.subject_code,
                         campus_code:  row.campus_code,
                         term_code:    row.term_code})
MATCH (s:Subject {subject_code: row.subject_code})
MATCH (c:Campus  {campus_code:  row.campus_code})
MATCH (t:Term    {term_code:    row.term_code})
MATCH (lead:Staff {person_id:   row.lead_staff_id})
MERGE (o)-[:OF_SUBJECT]->(s)
MERGE (o)-[:AT_CAMPUS]->(c)
MERGE (o)-[:IN_TERM]->(t)
MERGE (lead)-[:LEADS]->(o);

// 6.6 SUBJECT_ENROLMENT + RESULT -> one [:TOOK] relationship per enrolment.
//     RESULT is a 1:1 dependant of SUBJECT_ENROLMENT (EER cardinality
//     1..1 : 0..1), so its attributes are folded onto the same relationship.
LOAD CSV WITH HEADERS FROM 'file:///subject_enrolment.csv' AS row
MATCH (s:Student {person_id: row.student_person_id})
MATCH (o:CourseOffering {subject_code: row.subject_code,
                         campus_code:  row.campus_code,
                         term_code:    row.term_code})
MERGE (s)-[r:TOOK]->(o)
SET r.enrolled_on = date(row.enrolled_on),
    r.status      = row.se_status;

LOAD CSV WITH HEADERS FROM 'file:///result.csv' AS row
MATCH (s:Student {person_id: row.student_person_id})
      -[r:TOOK]->
      (o:CourseOffering {subject_code: row.subject_code,
                         campus_code:  row.campus_code,
                         term_code:    row.term_code})
SET r.result_type        = row.result_type,
    r.grade_letter       = CASE WHEN row.grade_letter <> '' THEN row.grade_letter END,
    r.competency_outcome = CASE WHEN row.competency_outcome <> '' THEN row.competency_outcome END,
    r.awarded_date       = date(row.awarded_date);

// -----------------------------------------------------------------------------
// 7. DERIVED STRUCTURES (graph-native optimisations; source data unchanged)
// -----------------------------------------------------------------------------

// 7.1 Term sequence: [:NEXT_TERM] links each term to the following one, so
//     "previous / next term" questions are one hop instead of date arithmetic.
MATCH (t:Term)
WITH t ORDER BY t.academic_year, t.term_number
WITH collect(t) AS terms
UNWIND range(0, size(terms) - 2) AS i
WITH terms[i] AS this_term, terms[i + 1] AS next_term
MERGE (this_term)-[:NEXT_TERM]->(next_term);

// 7.2 Materialised invoice balance (the derived attribute /total_amount of
//     the EER model). Lines and payments remain the source of truth; the
//     three properties are recomputed by re-running this statement.
MATCH (i:Invoice)
OPTIONAL MATCH (i)-[:HAS_LINE]->(l:InvoiceLine)
WITH i, sum(l.line_amount) AS total
OPTIONAL MATCH (p:Payment)-[:SETTLES]->(i)
WITH i, total, sum(p.amount_paid) AS paid
SET i.total_amount  = total,
    i.amount_paid   = paid,
    i.balance_owing = total - paid;


// -----------------------------------------------------------------------------
// 8. MIGRATION VALIDATION - every statement should return the value shown
// -----------------------------------------------------------------------------

// 8.1 Node counts by label (expected: Person 42, Student 24, Staff 10,
//     Guardian 11, Subject 8, QCAASubject 4, VETUnit 4, Campus 5, Term 4,
//     Program 4, CourseOffering 32, Enrolment 26, Qualification 2,
//     HostEmployer 8, Placement 11, PlacementAgreement 11,
//     BillingAccount 22, Invoice 42, InvoiceLine 73, Payment 33)
MATCH (n)
UNWIND labels(n) AS label
RETURN label, count(*) AS nodes
ORDER BY label;

// 8.2 Relationship counts by type (expected total 633: AT_CAMPUS 58, AWARDED 2,
//     BILLED_TO 42, FOR_TERM 42, GOVERNED_BY 11, GUARDIAN_OF 16,
//     HAS_ENROLMENT 26, HAS_LINE 73, HOLDS_ACCOUNT 22, HOSTED_BY 11,
//     INCLUDES 12, IN_PROGRAM 26, IN_TERM 32, LEADS 32, NEXT_TERM 3,
//     OF_SUBJECT 32, RAISED_FOR 42, REQUIRES 3, SETTLES 33, SUPERVISES 9,
//     TOOK 95, UNDERTAKES 11)
MATCH ()-[r]->()
RETURN type(r) AS relationship, count(*) AS relationships
ORDER BY relationship;

// 8.3 Every relational row accounted for: TOOK relationships carrying a
//     result must equal the 71 RESULT rows; all TOOK must equal the 95
//     SUBJECT_ENROLMENT rows.
MATCH ()-[r:TOOK]->()
RETURN count(r) AS subject_enrolments,
       count(r.result_type) AS results;

// 8.4 Categorisation integrity (Assessment 2 feedback). Constraint
//     holds_account_unique already rejects a second holder; this check also
//     confirms that every account HAS a holder and that each holder is one of
//     the category's three superclasses. Expected: no rows.
MATCH (b:BillingAccount)
OPTIONAL MATCH (h)-[:HOLDS_ACCOUNT]->(b)
WITH b, collect(h) AS holders
WHERE size(holders) <> 1
   OR NOT all(x IN holders WHERE x:Student OR x:Guardian OR x:HostEmployer)
RETURN b.account_no AS account_no, size(holders) AS holders;

// 8.5 Specialisation integrity: SUBJECT is disjoint and total.
//     Expected: overlapping_subjects 0, unspecialised_subjects 0.
MATCH (s:Subject)
RETURN sum(CASE WHEN s:QCAASubject AND s:VETUnit THEN 1 ELSE 0 END)      AS overlapping_subjects,
       sum(CASE WHEN NOT (s:QCAASubject OR s:VETUnit) THEN 1 ELSE 0 END) AS unspecialised_subjects;

// 8.6 PERSON specialisation is total: every person holds at least one role.
//     Expected: roleless_people 0. Multi-role people are allowed (overlapping).
MATCH (p:Person)
RETURN sum(CASE WHEN NOT (p:Student OR p:Staff OR p:Guardian) THEN 1 ELSE 0 END) AS roleless_people,
       sum(CASE WHEN size([l IN labels(p) WHERE l IN ['Student', 'Staff', 'Guardian']]) > 1
                THEN 1 ELSE 0 END)                                              AS multi_role_people;

// 8.7 Structural integrity - the graph equivalent of NOT NULL, single-valued
//     foreign keys. Every reified node must have exactly one of each mandatory
//     relationship, and an enrolment at most one qualification (the relational
//     UNIQUE (student_person_id, program_code) on QUALIFICATION).
//     Expected: one row, violations = 0. (Neo4j 5.23+ / Cypher 25 also shows a
//     "Feature deprecated" notice for this CALL form; on those versions it can be
//     written CALL () { ... } - the result is the same.)
CALL {
  MATCH (o:CourseOffering)
  WHERE COUNT { (o)-[:OF_SUBJECT]->() } <> 1 OR COUNT { (o)-[:AT_CAMPUS]->() } <> 1
     OR COUNT { (o)-[:IN_TERM]->() } <> 1    OR COUNT { (o)<-[:LEADS]-() } <> 1
  RETURN count(o) AS bad
  UNION ALL
  MATCH (e:Enrolment)
  WHERE COUNT { (e)<-[:HAS_ENROLMENT]-() } <> 1 OR COUNT { (e)-[:IN_PROGRAM]->() } <> 1
     OR COUNT { (e)-[:AT_CAMPUS]->() } <> 1     OR COUNT { (e)-[:AWARDED]->() } > 1
  RETURN count(e) AS bad
  UNION ALL
  MATCH (p:Placement)
  WHERE COUNT { (p)<-[:UNDERTAKES]-() } <> 1 OR COUNT { (p)-[:HOSTED_BY]->() } <> 1
     OR COUNT { (p)-[:GOVERNED_BY]->() } <> 1
  RETURN count(p) AS bad
  UNION ALL
  MATCH (i:Invoice)
  WHERE COUNT { (i)-[:BILLED_TO]->() } <> 1 OR COUNT { (i)-[:RAISED_FOR]->() } <> 1
     OR COUNT { (i)-[:FOR_TERM]->() } <> 1
  RETURN count(i) AS bad
  UNION ALL
  MATCH (x)
  WHERE (x:InvoiceLine AND COUNT { (x)<-[:HAS_LINE]-() } <> 1)
     OR (x:Payment AND COUNT { (x)-[:SETTLES]->() } <> 1)
  RETURN count(x) AS bad
}
RETURN sum(bad) AS violations;
