LOAD CSV WITH HEADERS FROM 'file:///person_phone.csv' AS row
MATCH (p:Person {person_id: row.person_id}), (ph:PhoneNumber {person_id: row.person_id, phone_number: row.phone_number})

LOAD CSV WITH HEADERS FROM 'file:///guardianship.csv' AS row
MATCH (g:Guardian {person_id: row.guardian_person_id}), (s:Student {person_id: row.student_person_id})
MERGE (g)-[r:GUARDIAN_OF]->(s)
SET r.relationship_type = row.relationship_type, r.is_primary_contact = row.is_primary_contact;

LOAD CSV WITH HEADERS FROM 'file:///staff.csv' AS row
WITH row WHERE row.supervisor_person_id <> ''
MATCH (b:Staff {person_id: row.supervisor_person_id}), (s:Staff {person_id: row.person_id})
MERGE (b)-[:SUPERVISES]->(s);

LOAD CSV WITH HEADERS FROM 'file:///program_subject.csv' AS row
MATCH (p:Program {program_code: row.program_code}), (s:Subject {subject_code: row.subject_code})

SET r.is_core = row.is_core, r.sequence_no = row.sequence_no;

LOAD CSV WITH HEADERS FROM 'file:///subject_prerequisite.csv' AS row

LOAD CSV WITH HEADERS FROM 'file:///course_offering.csv' AS row
MATCH (o:CourseOffering {subject_code: row.subject_code, campus_code: row.campus_code, term_code: row.term_code})
MATCH (s:Subject {subject_code: row.subject_code}), (c:Campus {campus_code: row.campus_code}),
      (t:Term {term_code: row.term_code}), (st:Staff {person_id: row.lead_staff_id})
MERGE (o)-[:OF_SUBJECT]->(s)
MERGE (o)-[:AT_CAMPUS]->(c)
MERGE (o)-[:IN_TERM]->(t)
MERGE (st)-[:LEADS]->(o);

LOAD CSV WITH HEADERS FROM 'file:///subject_enrolment.csv' AS row

MERGE (se)-[:FOR_OFFERING]->(o);
