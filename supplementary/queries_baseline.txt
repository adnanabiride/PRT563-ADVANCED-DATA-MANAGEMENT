// Q1-A  ASSESSMENT CONFLICT OF INTEREST (baseline)
MATCH (st:Staff)-[:IS_A]->(sp:Person)<-[:IS_A]-(gd:Guardian)-[g:GUARDIAN_OF]->(child:Student)
      -[:IS_A]->(cp:Person)
OPTIONAL MATCH (child)-[:HAS_SUBJECT_ENROLMENT]->(se:SubjectEnrolment)
               -[:FOR_OFFERING]->(o:CourseOffering)<-[:LEADS]-(st),
               (o)-[:OF_SUBJECT]->(subj:Subject),
               (o)-[:IN_TERM]->(term:Term)
OPTIONAL MATCH (se)-[:HAS_RESULT]->(r:Result)
RETURN sp.preferred_name + ' ' + sp.family_name   AS staff_member,
       st.position_title                           AS position,
       cp.preferred_name + ' ' + cp.family_name   AS dependant,
       g.relationship_type                         AS relationship,
       CASE WHEN o IS NULL THEN 'No conflict' ELSE 'CONFLICT' END AS finding,
       subj.subject_code + ' ' + subj.subject_title AS subject,
       o.campus_code                               AS campus,
       term.term_code                              AS term,
       CASE WHEN o IS NULL THEN null
            WHEN r IS NULL THEN 'Not yet assessed - reassign assessor now'
            ELSE 'Result awarded ' + r.awarded_date + ' - independent re-mark'
       END                                         AS action,
       COLLECT {
         MATCH (alt:Staff)-[:LEADS]->(:CourseOffering)-[:OF_SUBJECT]->(subj),
               (alt)-[:IS_A]->(ap:Person)
         WHERE alt <> st
           AND NOT EXISTS { (ap)<-[:IS_A]-(:Guardian)-[:GUARDIAN_OF]->(child) }
           AND date(alt.blue_card_expiry) >= date(term.end_date)
         RETURN DISTINCT ap.preferred_name + ' ' + ap.family_name
       }                                           AS eligible_assessors
ORDER BY finding, staff_member;


// Q2-A  BLUE CARD COMPLIANCE SWEEP FOR NEXT TERM (baseline)
//       no [:NEXT_TERM]: the next term is found by sorting on converted dates
WITH date('2026-09-14') AS as_at
MATCH (cur:Term)
WHERE date(cur.start_date) <= as_at <= date(cur.end_date)
MATCH (nxt:Term)
WHERE date(nxt.start_date) > date(cur.end_date)
WITH as_at, nxt ORDER BY date(nxt.start_date) LIMIT 1
MATCH (s:Staff)-[:IS_A]->(sp:Person)
WHERE date(s.blue_card_expiry) <= date(nxt.end_date)
MATCH chain = (top:Staff)-[:SUPERVISES*0..]->(s)
WHERE NOT EXISTS { ()-[:SUPERVISES]->(top) }
WITH as_at, nxt, s, sp,
     [x IN reverse(nodes(chain)) | head([(x)-[:IS_A]->(xp:Person) | xp])] AS upward,
     COLLECT { MATCH (s)-[:LEADS]->(o:CourseOffering)-[:IN_TERM]->(nxt)
               RETURN o.subject_code + '@' + o.campus_code } AS rostered,
     COLLECT { MATCH (s)-[:LEADS]->(o:CourseOffering)-[:IN_TERM]->(nxt),
                     (o)<-[:FOR_OFFERING]-(se:SubjectEnrolment)
                        <-[:HAS_SUBJECT_ENROLMENT]-(st:Student)
               WHERE se.se_status = 'ENROLLED'
               RETURN DISTINCT st.person_id } AS students
WITH as_at, nxt, s, sp, upward, rostered, students, upward[1] AS mgr
RETURN sp.preferred_name + ' ' + sp.family_name                    AS staff_member,
       s.position_title                                             AS position,
       date(s.blue_card_expiry)                                     AS blue_card_expiry,
       duration.inDays(as_at, date(s.blue_card_expiry)).days        AS days_remaining,
       CASE
         WHEN size(rostered) = 0 THEN 'RENEW - not rostered in ' + nxt.term_code
         WHEN date(s.blue_card_expiry) < date(nxt.start_date)
              THEN 'BLOCKER - lapses before ' + nxt.term_code + ' starts'
         ELSE 'BLOCKER - lapses during ' + nxt.term_code
       END                                                          AS risk,
       rostered                                                     AS offerings_next_term,
       size(students)                                               AS students_affected,
       reduce(chain_text = '', x IN upward |
              chain_text + CASE WHEN chain_text = '' THEN '' ELSE ' > ' END
                         + x.preferred_name)                        AS escalation_chain,
       coalesce(head([(mgr)-[:HAS_PHONE]->(ph:PhoneNumber)
                      WHERE ph.phone_type = 'MOBILE' | ph.phone_number]),
                head([(mgr)-[:HAS_PHONE]->(ph:PhoneNumber)
                      WHERE ph.phone_type = 'WORK' | ph.phone_number]),
                mgr.email)                                          AS manager_contact
ORDER BY risk, days_remaining;


// Q3-A  CAMPUS DEBT EXPOSURE (baseline)
//       no materialised balance: every invoice is re-totalled from its lines
//       and payments, with string amounts converted by toFloat()
WITH date('2026-09-14') AS as_at
MATCH (c:Campus)
OPTIONAL MATCH (c)<-[:AT_CAMPUS]-(:Enrolment {enrolment_status: 'ACTIVE'})
               <-[:HAS_ENROLMENT]-(s:Student)<-[:RAISED_FOR]-(i:Invoice)
WITH DISTINCT as_at, c, s, i,
     reduce(t = 0.0, a IN [(i)-[:HAS_LINE]->(l:InvoiceLine) | toFloat(l.line_amount)] | t + a)
   - reduce(t = 0.0, a IN [(i)<-[:SETTLES]-(p:Payment) | toFloat(p.amount_paid)] | t + a)
       AS balance
WITH as_at, c,
     collect(CASE WHEN balance > 0 THEN {due: date(i.due_date), balance: balance} END) AS open_invoices,
     count(DISTINCT CASE WHEN balance > 0 THEN s END) AS debtors
WITH as_at, c, open_invoices, debtors,
     reduce(total = 0.0, x IN open_invoices | total + x.balance) AS debt,
     reduce(aged = 0.0, x IN open_invoices |
            aged + CASE WHEN duration.inDays(x.due, as_at).days > 90
                        THEN x.balance ELSE 0.0 END)             AS aged_90
WITH collect({campus: c.campus_name, invoices: size(open_invoices),
              debtors: debtors, debt: debt, aged_90: aged_90})   AS campuses
WITH campuses,
     reduce(total = 0.0, x IN campuses | total + x.debt) / size(campuses) AS school_average
UNWIND campuses AS row
RETURN row.campus                                         AS campus,
       row.invoices                                       AS open_invoices,
       row.debtors                                        AS debtor_students,
       round(row.debt, 2)                                 AS balance_owing,
       round(row.debt - school_average, 2)                AS variance_vs_average,
       CASE WHEN row.debt = 0 THEN 0.0
            ELSE round(100.0 * row.aged_90 / row.debt, 1) END AS pct_over_90_days,
       CASE WHEN row.debt > school_average THEN 'ABOVE AVERAGE - escalate'
            ELSE 'within average' END                     AS flag,
       round(school_average, 2)                           AS school_average
ORDER BY balance_owing DESC;


// Q4-A  PREREQUISITE BREACHES AND THEIR ROOT CAUSE (baseline)
MATCH (s:Student)-[:HAS_SUBJECT_ENROLMENT]->(se:SubjectEnrolment)
      -[:FOR_OFFERING]->(o:CourseOffering)-[:OF_SUBJECT]->(subj:Subject),
      (o)-[:IN_TERM]->(term:Term)
WHERE se.se_status <> 'WITHDRAWN'
MATCH (subj)-[:REQUIRES]->+(pre:Subject)
WITH DISTINCT s, se, subj, term, pre
WITH s, se, subj, term, pre,
     [(s)-[:HAS_SUBJECT_ENROLMENT]->(se2:SubjectEnrolment)-[:FOR_OFFERING]->(o2:CourseOffering)
        -[:OF_SUBJECT]->(pre)
        | {term:    o2.term_code,
           start:   head([(o2)-[:IN_TERM]->(x:Term) | date(x.start_date)]),
           outcome: coalesce(head([(se2)-[:HAS_RESULT]->(r2:Result)
                                   | coalesce(r2.grade_letter, r2.competency_outcome)]),
                             se2.se_status)}]
       AS attempts
WHERE none(a IN attempts WHERE a.start < date(term.start_date)
                           AND a.outcome IN ['A', 'B', 'C'])
OPTIONAL MATCH (s)-[:HAS_ENROLMENT]->(:Enrolment {enrolment_status: 'ACTIVE'})
               -[:IN_PROGRAM]->(prog:Program)
MATCH (s)-[:IS_A]->(sp:Person)
RETURN s.student_no                                   AS student_no,
       sp.preferred_name + ' ' + sp.family_name       AS student,
       prog.program_code                              AS program,
       subj.subject_code                              AS subject,
       term.term_code                                 AS term,
       se.se_status                                   AS status,
       pre.subject_code                               AS unmet_prerequisite,
       CASE size(attempts)
         WHEN 0 THEN 'never attempted'
         ELSE reduce(h = '', a IN attempts |
                     h + CASE WHEN h = '' THEN '' ELSE ', ' END
                       + a.term + ' ' + a.outcome)
       END                                            AS prerequisite_history,
       CASE
         WHEN prog IS NULL                               THEN 'No active enrolment'
         WHEN NOT EXISTS { (prog)-[:INCLUDES]->(subj) }  THEN 'Enrolled outside program'
         WHEN NOT EXISTS { (prog)-[:INCLUDES]->(pre) }   THEN 'Program design gap'
         ELSE 'Progression control failure'
       END                                            AS root_cause
ORDER BY root_cause, program, student_no, subject;
